% UTILIDADES PARA FIGURAS
% Ejemplo de uso:
% \captura{recursos/capturas/pregunta-1/resultado.png}{Descripción de la evidencia.}{fig:resultado}
\newcommand{\captura}[3]{%
  \begin{figure}[H]
    \centering
    \includegraphics[width=0.95\linewidth]{#1}
    \caption{#2}\label{#3}
  \end{figure}
}

% El archivo puede ser .pdf para conservar diagramas vectoriales.
\newcommand{\diagrama}[3]{%
  \begin{figure}[H]
    \centering
    \includegraphics[width=0.85\linewidth]{#1}
    \caption{#2}\label{#3}
  \end{figure}
}
